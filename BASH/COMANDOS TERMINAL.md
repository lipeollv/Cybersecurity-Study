_Navegação_
/ = raiz do sistema;
/home = onde ficam os arquivos pessoais;
/etc = onde ficam as configurações do sistema;
/bin e /usr/bin = onde ficam os programas;
/var = onde ficam os logs;

_Comandos de pastas_
pwd (print working directory) = "onde eu estou?"/ mostra o caminho completo da pasta atual;

ls (list) = lista os arquivos e pastas;
	ls -la = mostra detalhes de permissões;
	-a = mostra arquivos ocultos (que começam com .);

cd (change directory) = "quero ir pra lá.";
	cd .. = volta uma pasta para trás;
	cd ~ = volta direto para pasta de usuário;

_Gerenciador de Tarefas_
htop = gerenciador de tarefas;
ps aux | grep = filtro. vai mostrar exatamente o PID de um programa específico.


PID = ID de um processo;
USER = quem rodou;
COMMAND = nome do programa;
